 25class WeatherAppMain:
 26    """This class is used for prompting the user to enter
 27    in an address to look up specific weather data about.
 28    It uses the GeoLocation class to then search for the
 29    address and get the location coordinates and pass it to
 30    the WeatherData class
 31    """
 32
 33    def __init__(self) -> None:
 34        """Initializes the Geopy module and the prompt
 35        class from Rich module
 36        """
 37        self.geo_loc = GeoLocation()
 38        self.prompt = Prompt()
 39
 40    def lookup(self, location: str) -> None:
 41        """Function  to pass in the address that was
 42        supplied by the user and sets the GeoLocation
 43        lat and lon values to those found values
 44
 45        Args:
 46            location (str): User entered address
 47        """
 48        self.geo_loc.lat_lon_lookup(location)
 49
 50    @property
 51    def loc(self) -> Tuple[str, str]:
 52        """Getter function to access the tuple that
 53        contains the string representations of the lat
 54        and lon
 55
 56        Returns:
 57            Tuple[str, str]: Tuple containing two strings lat and
 58            lon
 59        """
 60        return self.geo_loc.user_loc
 61
 62    def user_input(self) -> str:
 63        """Function for getting user input for an address
 64        using the Prompt class from Rich module
 65
 66        Returns:
 67            str: User entered address
 68        """
 69        location: str = \
 70            self.prompt.ask(
 71                "[chartreuse1]Enter a location separated by a ',' "
 72                "[/chartreuse1]")
 73        return location
 74
 75    @staticmethod
 76    def main() -> None:
 77        """Main function that handles all the calls for this
 78        class and the WeatherData class
 79        """
 80        wam = WeatherAppMain()

118    def convert_to_datetime(self, utc_num: float, frmt_opt: int) -> Any:
119        """Converts a UTC timestamp to a human-readable format
120        using datetime. Then formats it for specific values within
121        the JSON file
122
123        Args:
124            utc_num (float): UTC timestamp from JSON file
125            frmt_opt (int): Formatting option for specific JSON values
126
127        Returns:
128            Any: Formatted string depending on the option that was
129            used ex: Tues 1 March, 2024 7:34 PM MST-000 for 0 and
130            7:34 PM MST-000 for anything else
131        """
132        if frmt_opt == 0:
133            fmt = '%a %-d %B, %Y %-I:%M %p %Z%z'
134        else:
135            fmt = '%-I:%M %p %Z%z'
136        utc_dt = datetime.fromtimestamp(timestamp=utc_num, tz=timezone.utc)
137        loc_tz = pytz.timezone(self.api.timezone_loc)
138        loc_dt = loc_tz.normalize(utc_dt.astimezone(loc_tz))
139        return loc_dt.strftime(fmt)

181    def current_weather_data(self) -> None:  # noqa: C901
182        """Process the data returned from WeatherAPI that
183        stores the JSON values. As well as concatenates a
184        string of all the different data values
185
186        Raises:
187            ValueError: All value errors are just for checking
188            if a value is in there and flagging the program if
189            it is not
190        """
191        curr_w = self.api.current_weather
192
193        if curr_w.get('dt') is not None:
194            curr_time = \
195                self.convert_to_datetime(float(str(curr_w.get('dt'))), 0)
196        else:
197            raise ValueError('Date Time is Undefined')
198
199        if curr_w.get('sunrise') is not None:
200            curr_sunrise = \
201                self.convert_to_datetime(float(str(curr_w.get('sunrise'))), 1)
202        else:
203            raise ValueError('Sunrise is Undefined')
204
205        if curr_w.get('sunset') is not None:
206            curr_sunset = \
207                self.convert_to_datetime(float(str(curr_w.get('sunset'))), 1)
208        else:
209            raise ValueError('Sunset is Undefined')  # pragma: no cover
210
211        subtitle = f"[b]Current Weather for {self.location}[/]"
212        title = f"[b]{curr_time}[/]"
213        curr_w_output = f"[b]Sunrise:[/] {curr_sunrise}\n"
214        curr_w_output += f"[b]Sunset:[/] {curr_sunset}\n"
215
216        description: List[Dict[str, Any]]
217        weather_data: Optional[List[Dict[str, Any]]] = curr_w.get('weather')
218        if weather_data is not None:
219            description = weather_data
220            for desc in description:
221                main = desc.get('main')
222                main_desc = desc.get('description')
223                if main and main_desc:
224                    curr_w_output += \
225                        f"[b]Description:[/] {main} - {main_desc}\n"
226        else:
227            raise ValueError('Empty Weather Data List')  # pragma: no cover
228
229        curr_w_output += \
230            f"[b]Temperature:[/] {curr_w.get('temp')}\N{DEGREE SIGN}F " \
231            f"[b]Feels like:[/] {curr_w.get('feels_like')}\N{DEGREE SIGN}F\n"
232        curr_w_output += f"[b]Humidity:[/] {curr_w.get('humidity')}%\n"
233        curr_w_output += f"[b]Cloudiness:[/] {curr_w.get('clouds')}%\n"
234        curr_w_output += f"[b]UV Index:[/] {curr_w.get('uvi')}\n"
235
236        v_str: str | None = curr_w.get('visibility')
237        if v_str is not None:
238            v_float: float = float(v_str)
239            curr_w_output += (
240                f"[b]Visibility:[/] "
241                f"{(v_float * 3.28):.2f} "
242                f"Foot/Feet\n"
243            )
244        else:
245            raise ValueError('Visibility is Undefined')
246
247        curr_w_output += f"[b]Wind Speed:[/] {curr_w.get('wind_speed')}mph\n"
248
249        if curr_w.get('wind_gust') is not None:
250            curr_w_output += \
251                f"[b]Wind Gusts:[/] {curr_w.get('wind_gust')}mph\n"
252
253        curr_rain: Dict[str, Any] | None = curr_w.get('rain')
254        if curr_rain is not None:
255            curr_w_output += \
256                f"[b]Current Precipitation:[/] " \
257                f"{float(str(curr_rain.get('1h'))) / 25.4}\n" \
258                # pragma: no cover
259
260        curr_snow: Dict[str, Any] | None = curr_w.get('snow')
261        if curr_snow is not None:
262            curr_w_output += \
263                f"[b]Current Snow Accumulation:[/] " \
264                f"{float(str(curr_snow.get('1h'))) / 25.4}\n" \
265                # pragma: no cover
266
267        self.print_panel(curr_w_output, title, subtitle)

286                        1

288
289            if dw.get('sunset') is not None:
290                sst = \
291                    self.convert_to_datetime(
292                        float(str(dw.get('sunset'))),
293                        1
294                    )
295
296            if dw.get('moonrise') is not None:
297                mr = \
298                    self.convert_to_datetime(
299                        float(str(dw.get('moonrise'))),
300                        1
301                    )
302
303            if dw.get('moonset') is not None:
304                mst = \
305                    self.convert_to_datetime(
306                        float(str(dw.get('moonset'))),
307                        1
308                    )
309
310            title = f"[b]{dt}[/]"

319            dw_output += f"[b]Moon Phase:[/] {dw.get('moon_phase')}\n"
320            dw_output += (
321                f"[b]Daily Temp[/]\n[b]Morning Temp:[/] "
322                f"{dw['temp'].get('morn')}"
323                f"\N{DEGREE SIGN}F [b]Day Temp:[/] {dw['temp'].get('day')}"
324                f"\N{DEGREE SIGN}F [b]Evening Temp:[/] "

326                f"\N{DEGREE SIGN}F [b]Night Temp:[/] "
327                f"{dw['temp'].get('night')}"
328                f"\N{DEGREE SIGN}F\n [b]Min Daily Temp:[/] "
329                f"{dw['temp'].get('min')}"
330                f"\N{DEGREE SIGN}F [b]Max Daily Temp:[/] "
331                f"{dw['temp'].get('max')}"
332                f"\N{DEGREE SIGN}F\n"
333            )
334            dw_output += (
335                f"[b]Daily Feels Like[/]\n[b]Morning Temp:[/] "
336                f"{dw['feels_like'].get('morn')}\N{DEGREE SIGN}F "
337                f"[b]Day Temp:[/] "
338                f"{dw['feels_like'].get('day')}\N{DEGREE SIGN}F "
339                f"[b]Evening Temp:[/] "
340                f"{dw['feels_like'].get('eve')}\N{DEGREE SIGN}F"
341                f" [b]Night Temp:[/] {dw['feels_like'].get('night')}"
342                f"\N{DEGREE SIGN}F\n"
343            )
344            dw_output += f"[b]Humidity:[/] {dw.get('humidity')}%\n"
345            dw_output += f"[b]Wind Speed:[/] {dw.get('wind_speed')}mph\n"
346
347            if dw['wind_gust']:
348                dw_output += f"[b]Wind Gusts:[/] {dw.get('wind_gust')}mph\n"
349
350            dw_output += \
351                f"[b]Wind Direction:[/] " \
352                f"{dw.get('wind_deg')}\N{DEGREE SIGN}\n"
353            dw_output += f"[b]Cloudiness:[/] {dw.get('clouds')}%\n"
354            dw_output += f"[b]UV Index:[/] {dw.get('uvi')}\n"
355
356            pop: float | None = dw.get('pop')
357            if pop is not None:
358                dw_output += \
359                    f"[b]Probability of Precipitation:[/] " \
360                    f"{pop * 100}%\n"
361
362            if dw.get('rain') is not None:
363                dw_output += \
364                    f"[b]Preciptation Volume:[/] " \
365                    f"{dw.get('rain')} inch/hour\n"
366
367            if dw.get('snow') is not None:
368                dw_output += \
369                    f"[b]Snow Accumulation:[/] {dw.get('snow')} inch/hour\n" \
370                    # pragma: no cover
371
372            description: List[Dict[str, Any]]
373            weather_data: Optional[List[Dict[str, Any]]] = dw.get('weather')
374            if weather_data is not None:
375                description = weather_data
376                for desc in description:
377                    main = desc.get('main')
378                    main_desc = desc.get('description')
379                    if main and main_desc:
380                        dw_output += \

382            else:
383                raise ValueError('Empyt Weather Data List'
384                                 )  # pragma: no cover
385
386            self.print_panel(dw_output, title, subtitle)

388    def weather_alerts(self) -> None:
389        """This does the same as the previous function
390        that gets a list of dictionary values and concatenates
391        it into a string to be output
392        """
393        alerts = self.api.alerts
394
395        if alerts:
396            for aa in alerts:
397                title = f"[b]Advisory Alerts for {self.location}[/]\n"
398                aa_output = f"[b]Alert Sender:[/] {aa.get('sender_name')}\n"
399                aa_output += f"[b]Event Name:[/] {aa.get('event')}\n"
400                if aa.get('start') is not None:
401                    start_time = \
402                        self.convert_to_datetime(
403                            float(str(aa.get('start'))),
404                            0
405                        )
406
407                if aa.get('end') is not None:
408                    end_time = \
409                        self.convert_to_datetime(
410                            float(str(aa.get('end'))),
411                            0
412                        )
413
414                aa_output += f"[b]Start Time:[/] {start_time}\n"
415                aa_output += f"[b]End Time:[/] {end_time}\n"
416                aa_output += \
417                    f"[b]Alert Description:[/] {aa.get('description')}\n"
418
419                self.print_panel(aa_output, title, None)
420        else:
421            print(f"No Weather Advisories for {self.location}")
